// 01 — Core Concepts
OOP Concepts
Four pillars of Object-Oriented Programming with real e-commerce examples.
🔒 Encapsulation Wrap data + behavior
Bundle data (fields) and methods that operate on that data into a single unit. Hide internal state, expose only what's needed.
// E-Commerce: BankCard hides card number, exposes only masked version
public class BankCard
{
    private readonly string _cardNumber; // hidden
    private decimal _balance;

    public BankCard(string number, decimal balance)
    {
        _cardNumber = number;
        _balance = balance;
    }

    public string MaskedNumber => "**** **** **** " + _cardNumber[^4..];

    public bool Charge(decimal amount)
    {
        if (amount > _balance) return false;
        _balance -= amount;
        return true;
    }
}
🎭 Abstraction Show what matters
Expose only essential features. Hide implementation details. Use abstract classes or interfaces.
// Payment processor — caller doesn't care HOW payment is processed
public abstract class PaymentProcessor
{
    public abstract bool ProcessPayment(decimal amount);
    public abstract string ProviderName { get; }
}

public class StripeProcessor : PaymentProcessor
{
    public override string ProviderName => "Stripe";
    public override bool ProcessPayment(decimal amount)
    {
        // Stripe SDK logic hidden here
        Console.WriteLine($"Stripe charged {amount:C}");
        return true;
    }
}
🔀 Polymorphism Many forms
Same method name, different behavior. Two types: Compile-time (overloading) and Runtime (overriding).
// Runtime polymorphism — one loop handles all shipping types
public abstract class ShippingMethod
{
    public abstract decimal CalculateCost(double weightKg);
}

public class StandardShipping : ShippingMethod
{
    public override decimal CalculateCost(double w) => (decimal)(w * 2.5);
}

public class ExpressShipping : ShippingMethod
{
    public override decimal CalculateCost(double w) => (decimal)(w * 6.0 + 10);
}

// Usage — polymorphic dispatch
List<ShippingMethod> methods = [ new StandardShipping(), new ExpressShipping() ];
foreach (var method in methods)
    Console.WriteLine($"{method.GetType().Name}: {method.CalculateCost(3.5):C}");
🧬 Inheritance Reuse & extend
Child class inherits fields and methods from parent. Use for IS-A relationships. Avoid deep chains (>3 levels).
public class Product
{
    public string Name { get; set; }
    public decimal Price { get; set; }
    public virtual string GetDescription() => $"{Name} - {Price:C}";
}

public class DigitalProduct : Product
{
    public string DownloadUrl { get; set; }
    public override string GetDescription()
        => $"{base.GetDescription()} [Digital - {DownloadUrl}]";
}

public class PhysicalProduct : Product
{
    public double WeightKg { get; set; }
    public override string GetDescription()
        => $"{base.GetDescription()} [Weight: {WeightKg}kg]";
}
// 02 — Type System
C# Data Types
Value types live on stack. Reference types live on heap. Knowing the difference prevents bugs.
📦 Value Types Stack
Copied on assignment. int, double, bool, char, struct, enum, decimal.
int qty = 5;
int copy = qty;   // independent copy
copy = 10;
// qty is still 5

decimal price = 19.99m; // use decimal for money!
bool inStock = true;
char grade = 'A';
🔗 Reference Types Heap
Shared reference. class, string, array, interface, delegate.
var cart1 = new Cart();
var cart2 = cart1; // same object!
cart2.Items.Add("item");
// cart1.Items also has "item"

// Nullable reference type (C# 8+)
string? email = null; // explicit null
❓ Nullable Types
Allow value types to hold null. Use for optional database columns.
int? discountPercent = null; // no discount applied yet
DateTime? deliveredAt = null; // order not yet delivered

// Null coalescing
decimal finalDiscount = discountPercent ?? 0;

// Null conditional
string? city = order?.Address?.City;

// Null coalescing assignment (C# 8+)
discountPercent ??= 5; // assign 5 only if null
// 03 — Visibility
Access Modifiers
Control who can see and touch your code. Default to private, open up only what's needed.
public class OrderService
{
    // public — anyone can call this (API surface)
    public Order CreateOrder(int customerId) { ... }

    // private — only inside OrderService
    private void ValidateStock(List<CartItem> items) { ... }

    // protected — this class + subclasses
    protected virtual void ApplyDiscount(Order order) { ... }

    // internal — only within same assembly/project
    internal void SyncToWarehouse(Order order) { ... }

    // protected internal — assembly OR subclass
    protected internal DbContext Context { get; set; }
}
💡 Rule of thumb: Start with private. Only promote to public when something genuinely needs to be an API surface. This prevents tight coupling.
// 04 — Initialization
Constructors
Special methods that run when an object is created. Guarantee valid state from the start.
Constructor Types
public class Order
{
    public int Id { get; }
    public string CustomerEmail { get; }
    public List<OrderItem> Items { get; }
    public DateTime CreatedAt { get; }

    // 1. Default constructor
    public Order()
    {
        Items = new List<OrderItem>();
        CreatedAt = DateTime.UtcNow;
    }

    // 2. Parameterized constructor — enforces required data
    public Order(int id, string customerEmail) : this()
    {
        Id = id > 0 ? id : throw new ArgumentException("Invalid ID");
        CustomerEmail = customerEmail ?? throw new ArgumentNullException(nameof(customerEmail));
    }

    // 3. Static constructor — runs once for the class
    static Order() => Console.WriteLine("Order type initialized");
}

// Usage
var order = new Order(1, "user@example.com");
📌 Primary Constructor (C# 12): public class Order(int id, string email) — shorthand, params available throughout the class.
// 05 — Reuse
Inheritance
IS-A relationship. Prefer composition over inheritance for HAS-A relationships.
public abstract class Notification
{
    public string Recipient { get; set; }
    public string Message { get; set; }

    // Template Method Pattern — define skeleton, subclass fills in
    public void Send()
    {
        Validate();
        Deliver();
        LogSent();
    }

    protected abstract void Deliver(); // must override
    protected virtual void Validate() { } // optional override
    private void LogSent() => Console.WriteLine($"Sent to {Recipient}");
}

public class EmailNotification : Notification
{
    protected override void Deliver()
        => Console.WriteLine($"Email → {Recipient}: {Message}");
}

public class SmsNotification : Notification
{
    protected override void Deliver()
        => Console.WriteLine($"SMS → {Recipient}: {Message}");
}
⚠️ Avoid deep inheritance chains. If you're 3+ levels deep, switch to composition + interfaces. Inheritance creates tight coupling that's hard to refactor.
// 06 — Data Access
Properties & Indexers
Properties are smart fields with getters and setters. Indexers let you use [] syntax on your class.
Properties
public class Product
{
    private decimal _price;

    // Full property with validation
    public decimal Price
    {
        get => _price;
        set => _price = value >= 0 ? value : throw new ArgumentException("Price must be ≥ 0");
    }

    // Auto property
    public string Name { get; set; }

    // Init-only (C# 9) — set at construction only
    public string Sku { get; init; }

    // Computed property — no backing field
    public bool IsExpensive => Price > 1000m;
}

// Object initializer with init-only
var p = new Product { Name = "Laptop", Sku = "LAP-001", Price = 1299m };
Indexer
public class ShoppingCart
{
    private Dictionary<string, CartItem> _items = new();

    // Indexer — access items like cart["sku"]
    public CartItem this[string sku]
    {
        get => _items.TryGetValue(sku, out var item) ? item : null;
        set => _items[sku] = value;
    }
}

// Usage
var cart = new ShoppingCart();
cart["LAP-001"] = new CartItem { Quantity = 2 };
var item = cart["LAP-001"];

Struct Value Type
// Good for: small, immutable data
public readonly struct Money
{
    public decimal Amount { get; }
    public string Currency { get; }

    public Money(decimal a, string c)
        => (Amount, Currency) = (a, c);

    public static Money operator +(Money a, Money b)
    {
        if (a.Currency != b.Currency)
            throw new InvalidOperationException();
        return new Money(a.Amount + b.Amount, a.Currency);
    }
}

var price = new Money(100m, "USD");
var tax   = new Money(18m,  "USD");
var total = price + tax; // 118 USD

// 08 — C# Keywords
Special Keywords
yield, record, partial — modern C# features used in real systems.
yield Lazy iteration
Returns elements one at a time. Never materializes the full list in memory. Perfect for large datasets.
// Streams 1 million orders without loading all into RAM
public static IEnumerable<Order> GetLargeOrders(IEnumerable<Order> orders)
{
    foreach (var order in orders)
    {
        if (order.Total > 500m)
            yield return order; // one at a time
    }
}

// Infinite sequence — no memory issue
public static IEnumerable<int> OrderIds(int start)
{
    while (true) yield return start++;
}
record Immutable data model
Value equality, immutability, non-destructive mutation with with. Perfect for DTOs and domain events.
// Record — equality by value, not reference
public record OrderCreatedEvent(
    int OrderId,
    string CustomerEmail,
    decimal Total,
    DateTime CreatedAt
);

var e1 = new OrderCreatedEvent(1, "a@b.com", 99m, DateTime.Now);
var e2 = e1 with { Total = 120m }; // new record, one field changed

// e1.Total is still 99m — original unchanged
Console.WriteLine(e1 == e2); // false — different Total

// 09 — Design Principles
SOLID Principles
5 rules that make code maintainable, extensible, and testable.
S
Single Responsibility
One class = one job. If you need "and" to describe a class's purpose, split it.
O
Open / Closed
Open for extension (add new classes), closed for modification (don't edit existing code).
L
Liskov Substitution
Subclasses must work wherever the parent class is expected. No surprises.
I
Interface Segregation
Many small interfaces beat one fat interface. Clients implement only what they need.
D
Dependency Inversion
Depend on abstractions (interfaces), not implementations. Pass IOrderRepository, not SqlOrderRepository.
SOLID in action — Order Processing
// S — each class has one job
public class OrderValidator  { ... }   // only validates
public class OrderPricer     { ... }   // only calculates price
public class OrderRepository { ... }   // only persists

// D — depend on abstraction
public interface IOrderRepository
{
    Task<Order> GetByIdAsync(int id);
    Task SaveAsync(Order order);
}

// OrderService takes interface — not concrete SQL class
public class OrderService

    public OrderService(IOrderRepository repo) => _repo = repo;

public interface IPaymentGateway { Task<bool> ChargeAsync(decimal amount); }
public class StripeGateway  : IPaymentGateway { ... }
public class RazorpayGateway: IPaymentGateway { ... } // new — no existing change

Strategy Swappable algorithms
Encapsulate algorithms. Switch at runtime. No if/else chains.
public interface IDiscountStrategy
{
    decimal Calculate(decimal originalPrice);
}

public class PercentageDiscount : IDiscountStrategy
{
    private readonly decimal _percent;
    public PercentageDiscount(decimal percent) => _percent = percent;
    public decimal Calculate(decimal price) => price * (1 - _percent / 100);
}

public class FlatDiscount : IDiscountStrategy
{
    private readonly decimal _amount;
    public FlatDiscount(decimal amount) => _amount = amount;
    public decimal Calculate(decimal price) => Math.Max(0, price - _amount);
}

public class CheckoutService
{
    private IDiscountStrategy _strategy = new PercentageDiscount(0);

    public void SetDiscount(IDiscountStrategy s) => _strategy = s;
    public decimal Checkout(decimal price) => _strategy.Calculate(price);
}

// Black Friday — swap strategy at runtime
var svc = new CheckoutService();
svc.SetDiscount(new PercentageDiscount(20));
Console.WriteLine(svc.Checkout(1000m)); // 800
Decorator Add behavior at runtime
Wrap an object to add new features without changing its code.
public interface IOrderRepository
{
    Task<Order> GetByIdAsync(int id);
}

// Base implementation
public class SqlOrderRepository : IOrderRepository
{
    public async Task<Order> GetByIdAsync(int id)
        => await db.FindAsync<Order>(id);
}

// Decorator — adds caching without touching SqlOrderRepository
public class CachedOrderRepository : IOrderRepository
{
    private readonly IOrderRepository _inner;
    private readonly IMemoryCache _cache;

    public CachedOrderRepository(IOrderRepository inner, IMemoryCache cache)
        => (_inner, _cache) = (inner, cache);

    public async Task<Order> GetByIdAsync(int id)
        => await _cache.GetOrCreateAsync($"order:{id}", _ => _inner.GetByIdAsync(id));
}

CQRS Separate reads and writes
Commands mutate state. Queries return data. Never both. Use with MediatR in .NET.
// Command — changes state, returns nothing or just ID
public record CreateOrderCommand(string CustomerEmail, List<CartItem> Items);

public class CreateOrderHandler : IRequestHandler<CreateOrderCommand, int>
{
    public async Task<int> Handle(CreateOrderCommand cmd, CancellationToken ct)
    {
        var order = new Order(cmd.CustomerEmail, cmd.Items);
        await _repo.SaveAsync(order, ct);
        return order.Id;
    }
}

// Query — only reads, no side effects
public record GetOrderQuery(int OrderId);

public class GetOrderHandler : IRequestHandler<GetOrderQuery, OrderDto>
{
    public async Task<OrderDto> Handle(GetOrderQuery q, CancellationToken ct)
        => await _readDb.Orders.Where(o => o.Id == q.OrderId)
                               .Select(o => new OrderDto(o))
                               .FirstOrDefaultAsync(ct);
}

Facade — simple interface over complex subsystem
// OrderFacade hides: inventory check, payment, shipping, notification
public class OrderFacade
{
    private readonly InventoryService _inventory;
    private readonly PaymentService _payment;
    private readonly ShippingService _shipping;
    private readonly NotificationService _notifications;

    public async Task<bool> PlaceOrderAsync(CartDto cart)
    {
        if (!await _inventory.ReserveAsync(cart.Items)) return false;
        if (!await _payment.ChargeAsync(cart.Total))
        {
            await _inventory.ReleaseAsync(cart.Items);
            return false;
        }
        await _shipping.ScheduleAsync(cart);
        await _notifications.SendConfirmationAsync(cart.CustomerEmail);
        return true;
    }
}

// Caller just does this:
var success = await _facade.PlaceOrderAsync(cart);

// Publisher — Order raises events
public class Order
{
    public event EventHandler<OrderStatusChangedArgs> StatusChanged;

    private OrderStatus _status;
    public OrderStatus Status
    {
        get => _status;
        set
        {
            _status = value;
            StatusChanged?.Invoke(this, new(this, value));
        }
    }
}

Hangfire — Background Jobs
Schedule tasks outside the HTTP request cycle. Fire-and-forget, delayed, recurring.
// Fire-and-forget — send confirmation email after order
