  • QWhat is Native AOT in .NET 10 and when should you use it?
    Native AOT compiles your application ahead-of-time to a native executable, eliminating the JIT compiler at runtime. Use it for microservices and serverless functions where cold-start latency and memory footprint matter most. Tradeoff: limited reflection support and longer build times.
  • QExplain the difference between Minimal APIs and Controller-based APIs.
    Minimal APIs are function-based, lower ceremony, and have better performance due to fewer middleware layers. Controller-based APIs offer conventions, filters, model binding, and are better for complex domain scenarios. Choose Minimal APIs for microservices and simple CRUD; controllers for complex domains with extensive cross-cutting concerns.
  • QHow does Output Cache differ from Response Cache in ASP.NET Core?
    Output Cache is server-side and stores the actual rendered output — more flexible, configurable per-endpoint, and supports cache invalidation. Response Cache sets HTTP headers to instruct clients/proxies to cache. Use Output Cache for server-controlled caching; Response Cache for CDN-level caching.
  • QWhat are primary constructors in C# 13?
    Primary constructors allow parameters to be declared directly in the class/struct declaration. They are in scope throughout the type body, reducing boilerplate for simple dependency injection patterns. They are particularly useful with records and value objects.
✅ Best Practice
Always register services with appropriate lifetimes: Singleton for stateless, Scoped for per-request (DB context), Transient for lightweight services.
✅ Performance
Use IAsyncEnumerable for streaming large datasets instead of loading all data into memory first.
⚠️ Avoid
Never inject DbContext as Singleton — it's not thread-safe. Always use Scoped or create a factory.
⚠️ Security
Never store secrets in appsettings.json. Use Azure Key Vault, environment variables, or .NET User Secrets in development.
EF Core Best Practices
  • Use AsNoTracking() for read-only queries — up to 30% faster
  • Split queries for collections to avoid cartesian explosion
  • Use Compiled Queries for hot-path performance
  • Configure indexes via Fluent API, not Data Annotations
  • Use value converters for complex types (JSON columns)
  • Enable query logging only in development

// EF Core - Repository Pattern + Spec
public class ProductRepository(AppDbContext ctx)
    : IProductRepository
{
    public async Task> GetActiveAsync(
        CancellationToken ct = default)
        => await ctx.Products
            .AsNoTracking()
            .Where(p => p.IsActive && !p.IsDeleted)
            .OrderBy(p => p.Name)
            .Select(p => new Product(p.Id, p.Name, p.Price))
            .ToListAsync(ct);

    // Compiled query for hot paths
    private static readonly Func>
        GetByIdQuery = EF.CompileAsyncQuery(
            (AppDbContext db, int id) =>
                db.Products.FirstOrDefault(p => p.Id == id));

    public Task GetByIdAsync(int id)
        => GetByIdQuery(ctx, id);
}
CSHARP · CQRS Command Handler
// CQRS Command + Handler with MediatR
public sealed record CreateOrderCommand(
    Guid CustomerId,
    IReadOnlyList Items) : IRequest>;

public sealed class CreateOrderCommandHandler(
    IOrderRepository orders,
    ICustomerRepository customers,
    IUnitOfWork uow,
    IPublisher publisher)
    : IRequestHandler>
{
    public async Task> Handle(
        CreateOrderCommand cmd,
        CancellationToken ct)
    {
        var customer = await customers.GetByIdAsync(cmd.CustomerId, ct);
        if (customer is null)
            return Result.Failure(CustomerErrors.NotFound);

        var order = Order.Create(customer.Id, cmd.Items);
        await orders.AddAsync(order, ct);
        await uow.SaveChangesAsync(ct);

        // Publish domain event
        await publisher.Publish(
            new OrderCreatedEvent(order.Id, customer.Email), ct);

        return Result.Success(order.Id);
    }
}

// MediatR Pipeline Behaviour for Validation
public sealed class ValidationBehaviour(
    IEnumerable> validators)
    : IPipelineBehavior
    where TRequest : IRequest
{
    public async Task Handle(TRequest request,
        RequestHandlerDelegate next, CancellationToken ct)
    {
        var ctx = new ValidationContext(request);
        var failures = validators
            .Select(v => v.Validate(ctx))
            .SelectMany(r => r.Errors)
            .Where(f => f is not null)
            .ToList();

        if (failures.Count != 0)
            throw new ValidationException(failures);

        return await next();
    }
}

name: CI/CD Production Pipeline

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

env:
  REGISTRY: myregistry.azurecr.io
  IMAGE_NAME: myapp-api

jobs:
  build-and-test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-dotnet@v4
        with:
          dotnet-version: '10.0.x'

      - name: Restore & Build
        run: dotnet build --configuration Release --no-restore

      - name: Run Tests
        run: |
          dotnet test --no-build \
            --configuration Release \
            --collect:"XPlat Code Coverage" \
            --results-directory ./coverage

      - name: Code Coverage Gate (80%+)
        uses: codecov/codecov-action@v4

  docker-push:
    needs: build-and-test
    runs-on: ubuntu-latest
    if: github.ref == 'refs/heads/main'
    steps:
      - uses: actions/checkout@v4

      - name: Login to ACR
        uses: docker/login-action@v3
        with:
          registry: ${{ env.REGISTRY }}
          username: ${{ secrets.ACR_USERNAME }}
          password: ${{ secrets.ACR_PASSWORD }}

      - name: Build & Push Docker Image
        uses: docker/build-push-action@v5
        with:
          push: true
          tags: ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}:${{ github.sha }}

  deploy-production:
    needs: docker-push
    runs-on: ubuntu-latest
    environment: production
    steps:
      - name: Deploy to AKS
        run: |
          helm upgrade --install myapp-api ./charts/myapp \
            --set image.tag=${{ github.sha }} \
            --set replicas=3 \
            --namespace production \
            --wait --timeout 5m

// Redis distributed caching pattern
public class CachedProductService(
    IProductRepository repo,
    IDistributedCache cache) : IProductService
{
    public async Task GetByIdAsync(int id, CancellationToken ct)
    {
        var key = $"product:{id}";
        var cached = await cache.GetStringAsync(key, ct);
        if (cached is not null)
            return JsonSerializer.Deserialize(cached);

        var product = await repo.GetByIdAsync(id, ct);
        if (product is not null)
            await cache.SetStringAsync(key,
                JsonSerializer.Serialize(product),
                new DistributedCacheEntryOptions
                {
                    AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(15),
                    SlidingExpiration = TimeSpan.FromMinutes(5)
                }, ct);
        return product;
    }
}

Performance Optimization Checklist
API Level
  • Response compression (Brotli > Gzip)
  • Output Caching with cache tags
  • Async all the way down — no .Result/.Wait()
  • ValueTask for hot-path allocations

Database Level
  • Covering indexes for frequent queries
  • Read replicas for reporting workloads
  • Pagination — never load unbounded sets
  • Batch inserts with ExecuteUpdateAsync
  • Connection pooling and keep-alive

  • Database sharding strategies
